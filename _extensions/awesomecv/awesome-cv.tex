%-------------------------------------------------------------------------------
%	PANDOC SUPPORT
%-------------------------------------------------------------------------------
% common.latex is required in every custom Quarto LaTeX template, and calls
% pandoc.tex, which in turn brings in syntax highlighting, tables.tex,
% graphics.tex, citations.tex, babel-lang.tex, tightlist.tex, biblio-config.tex
% and header-includes. Do not hand-copy any of those: they are Pandoc's to
% maintain, and a stale local copy silently drifts out of step with the reader.
$common.latex()$
$after-header-includes.latex()$
$hypersetup.latex()$

% Typography. font-settings.latex honours mainfont/sansfont/monofont and their
% *options and *fallback variants, so a user changes the CV's fonts the same way
% they would in any other Quarto pdf. fonts.latex is NOT called: it would load
% lmodern and unicode-math over the top of what the class has already set up.
$font-settings.latex()$

% The class does not read \rmfamily directly -- every style command goes through
% \bodyfont, \headerfont or \footerfont, which resolve to \sffamily -- so
% \setmainfont above has no effect until those are pointed at it. Awesome-CV's
% own defaults (Source Sans 3 for the body, Roboto for the header) stand
% untouched when nothing is specified. The light faces fall back to the regular
% weight, as an arbitrary mainfont is not guaranteed to have a {l} face.
$if(mainfont)$
\renewcommand*{\bodyfont}{\rmfamily}
\renewcommand*{\bodyfontlight}{\rmfamily}
\renewcommand*{\footerfont}{\rmfamily}
$endif$
$if(headerfont)$
% \setfontfamily redefines the family the class declared with \newfontfamily.
% \headerfont/\headerfontlight wrap \roboto, so overriding the family they wrap
% carries through to both. The name given here resolves against installed
% system fonts, rather than the class default of Roboto from the texmf tree.
\setfontfamily\roboto[$for(headerfontoptions)$$headerfontoptions$$sep$,$endfor$]{$headerfont$}
% \headerfontlight asks for the {l} face, which an arbitrary system font may not
% have; drop back to the upright face so the name does not silently vanish.
\renewcommand*{\headerfontlight}{\roboto\selectfont}
$endif$

% Four further partials are deliberately not called here:
%   fonts.latex  see above.
%   before-title.tex, title.tex, before-body.tex, toc.tex  a CV has no title
%     block or contents page; \makecvheader below plays that role, and the
%     class's \section does not write \addcontentsline, so a toc would be empty.

\begin{document}
